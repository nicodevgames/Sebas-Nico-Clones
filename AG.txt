import { HttpError } from '../../../../shared/http/errors';
import type { HubWalletFacade } from '../../../domain/ProviderAdapter';
import type { WalletOkResponse } from '../../../domain/walletProtocol';

/** Código estable de un error del Hub/ledger (o undefined si no es un HttpError). */
export const errorCode = (e: unknown): string | undefined => (e instanceof HttpError ? e.code : undefined);

/** Ids namespaced por tipo: los agregadores pueden reusar el mismo transaction_id para bet y win. */
export const txKey = (type: 'bet' | 'win' | 'refund' | 'rollback', providerTxId: string): string => `${type}:${providerTxId}`;

/**
 * Los agregadores no cierran rondas explícitamente y pueden mandar varios wins (o bets) en una misma ronda.
 * Si la ronda ya está liquidada, la operación va a una sub-ronda determinista `<ronda>#<tx>`; la reentrega
 * del mismo tx vuelve a caer en esa sub-ronda y la idempotencia del Hub devuelve la respuesta original.
 */
export async function withRoundFallback(
  roundId: string,
  providerTxId: string,
  op: (roundId: string) => Promise<WalletOkResponse>,
): Promise<WalletOkResponse> {
  try {
    return await op(roundId);
  } catch (e) {
    const code = errorCode(e);
    if (code !== 'ROUND_CLOSED' && code !== 'DUPLICATE') throw e;
    return op(`${roundId}#${providerTxId}`);
  }
}

export type ReversalType = 'bet' | 'win' | 'refund';

/**
 * Busca la transacción original de una reversión en el orden indicado (por tipo declarado o, si no, bet → win → refund).
 * Devuelve la clave namespaced encontrada o null.
 */
export async function findOriginal(wallet: HubWalletFacade, providerTxId: string, declared?: string): Promise<string | null> {
  const order: ReversalType[] =
    declared === 'win' ? ['win', 'bet', 'refund'] : declared === 'refund' ? ['refund', 'bet', 'win'] : ['bet', 'win', 'refund'];
  for (const t of order) {
    const key = txKey(t, providerTxId);
    if (await wallet.findTx(key)) return key;
  }
  return null;
}

/** Lee el cuerpo como texto acotado (los catálogos pueden ser grandes; nunca sin límite). */
export async function readLimited(res: Response, maxBytes: number): Promise<string> {
  const len = Number(res.headers.get('content-length') ?? '0');
  if (len > maxBytes) throw new Error(`Respuesta demasiado grande (${len} bytes)`);
  const buf = Buffer.from(await res.arrayBuffer());
  if (buf.length > maxBytes) throw new Error(`Respuesta demasiado grande (${buf.length} bytes)`);
  return buf.toString('utf8');
}

/** Error saliente hacia el proveedor (lanzamiento/catálogo): 502 con el detalle recortado y sin secretos. */
export function providerError(provider: string, message: string, secrets: string[] = []): HttpError {
  let msg = message.slice(0, 200);
  for (const s of secrets) if (s) msg = msg.split(s).join('[clave]');
  return new HttpError(502, 'PROVIDER_ERROR', `${provider}: ${msg}`);
}
